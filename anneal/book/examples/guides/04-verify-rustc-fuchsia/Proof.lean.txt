private theorem u64Val_eq_zeroExt (w : BitVec 64) :
    Spec.u64Val w = .scalar ⟨.i64, false, bvZeroExtend 128 w⟩ := by
  unfold Spec.u64Val
  apply congrArg (fun b => Val.scalar ⟨.i64, false, b⟩)
  apply bitVec_eq_of_toNat_eq
  rw [bvZeroExtend_128_toNat]
  exact Nat.mod_eq_of_lt (Nat.lt_trans w.isLt (by decide))

private theorem u64Val_of_truncate {bits : BitVec 128}
    (hTrunc : truncateBits .i64 bits = bits) :
    Spec.u64Val (bvExtractLsb 63 0 bits) = .scalar ⟨.i64, false, bits⟩ := by
  have hLt : bits.toNat < Nat.pow 2 64 := by
    have hEq := congrArg BitVec.toNat hTrunc
    rw [toNat_truncateBits] at hEq
    rw [← hEq]
    exact Nat.mod_lt _ (by decide)
  have hMod : Nat.mod bits.toNat (Nat.pow 2 64) = bits.toNat :=
    Nat.mod_eq_of_lt hLt
  unfold Spec.u64Val bvExtractLsb
  apply congrArg (fun b => Val.scalar ⟨.i64, false, b⟩)
  apply bitVec_eq_of_toNat_eq
  change Nat.mod (Nat.mod bits.toNat (Nat.pow 2 64)) (Nat.pow 2 128) =
    bits.toNat
  rw [hMod]
  exact Nat.mod_eq_of_lt (Nat.lt_trans hLt (by decide))

private theorem u64BitAndVal_eq (x y : BitVec 64) :
    u64BitAndVal x y = Spec.u64Val (x &&& y) := by
  unfold u64BitAndVal Spec.u64Val
  apply congrArg (fun b => Val.scalar ⟨.i64, false, b⟩)
  apply bitVec_eq_of_toNat_eq
  rfl

private theorem fin4_cases {p : Fin 4 → Prop}
    (h0 : p 0) (h1 : p 1) (h2 : p 2) (h3 : p 3) : ∀ w, p w :=
  Fin.cases h0
    (Fin.cases h1 (Fin.cases h2 (Fin.cases h3 (fun i => Fin.elim0 i))))

private theorem executes_mask0 (value : BitVec 64) :
    Executes Captured.body_0
      (mkState 0 0 (initLocalsLoop Captured.body_0
        [Spec.widthArg Captured.body_0 0, Spec.u64Val value]
        0 Captured.body_0.locals))
      (Spec.u64Val (value &&& Spec.widthMaskBits 0)) := by
  rw [u64Val_eq_zeroExt value]
  let v2 : Val := .scalar ⟨.i64, false, bvZeroExtend 128 value⟩
  let w0 := Spec.widthArg Captured.body_0 0
  let d0 : Val := .scalar ⟨.i8, false, BitVec.ofNat 128 0⟩
  change Executes Captured.body_0
    (mkState 0 0
      [.immediate none, .immediate (some w0), .immediate (some v2),
        .immediate none, .dead, .dead, .dead])
    (Spec.u64Val (value &&& Spec.widthMaskBits 0))
  refine .next (next := mkState 0 1
    [.immediate none, .immediate (some w0), .immediate (some v2),
      .immediate none, .dead, .dead, .dead]) rfl ?_
  refine .next (next := mkState 0 2
    [.immediate none, .immediate (some w0), .immediate (some v2),
      .immediate (some d0), .dead, .dead, .dead]) rfl ?_
  refine .next (next := mkState 5 0
    [.immediate none, .immediate (some w0), .immediate (some v2),
      .immediate none, .dead, .dead, .dead]) rfl ?_
  refine .next (next := mkState 5 1
    [.immediate none, .immediate (some w0), .immediate (some v2),
      .immediate none, .immediate none, .dead, .dead]) rfl ?_
  refine .next (next := mkState 5 2
    [.immediate none, .immediate (some w0), .immediate (some v2),
      .immediate none, .immediate (some v2), .dead, .dead])
    (step_u64Use value rfl rfl) ?_
  refine .next (next := mkState 5 3
    [.immediate (some (u64BitAndVal value (BitVec.ofNat 64 0xff))),
      .immediate (some w0), .immediate (some v2), .immediate none,
      .immediate none, .dead, .dead])
    (step_u64And value (BitVec.ofNat 64 0xff) rfl rfl) ?_
  rw [u64BitAndVal_eq]
  refine .next (next := mkState 5 4
    [.immediate (some (Spec.u64Val (value &&& Spec.widthMaskBits 0))),
      .immediate (some w0), .immediate (some v2), .immediate none, .dead,
      .dead, .dead]) rfl ?_
  refine .next (next := mkState 6 0
    [.immediate (some (Spec.u64Val (value &&& Spec.widthMaskBits 0))),
      .immediate (some w0), .immediate (some v2), .immediate none, .dead,
      .dead, .dead]) rfl ?_
  exact .ret rfl

private theorem executes_mask3 (value : BitVec 64) :
    Executes Captured.body_0
      (mkState 0 0 (initLocalsLoop Captured.body_0
        [Spec.widthArg Captured.body_0 3, Spec.u64Val value]
        0 Captured.body_0.locals))
      (Spec.u64Val (value &&& Spec.widthMaskBits 3)) := by
  rw [u64Val_eq_zeroExt value]
  let v2 : Val := .scalar ⟨.i64, false, bvZeroExtend 128 value⟩
  let w3 := Spec.widthArg Captured.body_0 3
  let d3 : Val := .scalar ⟨.i8, false, BitVec.ofNat 128 3⟩
  change Executes Captured.body_0
    (mkState 0 0
      [.immediate none, .immediate (some w3), .immediate (some v2),
        .immediate none, .dead, .dead, .dead])
    (Spec.u64Val (value &&& Spec.widthMaskBits 3))

  refine .next (next := mkState 0 2

    [.immediate none, .immediate (some w3), .immediate (some v2),
      .immediate none, .dead, .dead, .dead]) rfl ?_

    [.immediate (some v2), .immediate (some w3), .immediate (some v2),

    (step_u64Use value rfl rfl) ?_
  have hAll : Spec.u64Val (value &&& Spec.widthMaskBits 3) = v2 := by
    have hm : value &&& Spec.widthMaskBits 3 = value := BitVec.and_allOnes
    rw [hm, u64Val_eq_zeroExt]

  refine .next (next := mkState 6 0
    [.immediate (some v2), .immediate (some w3), .immediate (some v2),
      .immediate none, .dead, .dead, .dead]) rfl ?_
  exact .ret rfl

private theorem executes_mask (idx : Fin 4) (value : BitVec 64) :
    Executes Captured.body_0
      (mkState 0 0 (initLocalsLoop Captured.body_0
        [Spec.widthArg Captured.body_0 idx.val, Spec.u64Val value]
        0 Captured.body_0.locals))
      (Spec.u64Val (value &&& Spec.widthMaskBits idx.val)) := by
  revert idx value
  exact fin4_cases executes_mask0 executes_mask1 executes_mask2 executes_mask3

private theorem widthArg_valid (idx : Fin 4) (value : BitVec 64) :
    ValidEntryArgs Captured.body_0
      [Spec.widthArg Captured.body_0 idx.val, Spec.u64Val value] := by
  have hEnum :
      checkVal (Spec.widthArg Captured.body_0 idx.val)
        (match Captured.body_0.locals with
        | _ :: (.enum ety) :: _ => .enum ety
        | _ => .unit) = true := by
    revert idx
    exact fin4_cases rfl rfl rfl rfl
  rw [u64Val_eq_zeroExt value]
  unfold ValidEntryArgs
  rw [show (Captured.body_0.locals.drop 1).take Captured.body_0.argumentCount =
    [match Captured.body_0.locals with
      | _ :: (.enum ety) :: _ => .enum ety
      | _ => .unit,
     .int .i64 false] from rfl]
  change (checkVal (Spec.widthArg Captured.body_0 idx.val) _ &&
    (checkVal (.scalar ⟨.i64, false, bvZeroExtend 128 value⟩)
      (.int .i64 false) && true)) = true
  rw [hEnum, checkVal_u64_bvZeroExtend value]
  rfl

theorem verify : Expected.claim := by
  refine ⟨rfl, rfl, rfl, rfl, rfl, rfl, ?_, ?_⟩
  · intro id body hFound
    cases id with
    | succ _ => cases hFound
    | zero =>
      have hEq : Captured.body_0 = body := Option.some.inj hFound
      subst body
      refine bodyVerified_of_safe rfl rfl (fun args hArgs _ => ?_)
      unfold ValidEntryArgs at hArgs
      change checkArgsMatch args
        [match Captured.body_0.locals with
          | _ :: (.enum ety) :: _ => .enum ety
          | _ => .unit,
         .int .i64 false] = true at hArgs
      let ⟨a0, rest0, h0, ha0, h1⟩ := checkArgsMatch_cons_iff.mp hArgs
      let ⟨a1, rest1, h1e, ha1, h2⟩ := checkArgsMatch_cons_iff.mp h1
      have hNil : rest1 = [] := checkArgsMatch_nil_iff.mp h2
      subst h0; subst h1e; subst hNil
      let ⟨_, ⟨idx, hIdx⟩⟩ := checkVal_enum_iff.mp ha0
      let ⟨b1, hb1, ht1⟩ := checkVal_int_iff.mp ha1
      subst hIdx; subst hb1
      have hArg0 :
          .enumVal _ idx.val = Spec.widthArg Captured.body_0 idx.val :=
        rfl
      rw [← u64Val_of_truncate ht1]
      exact executes_safe (executes_mask idx _)
  · refine ⟨Captured.body_0, rfl, fun idx value => ?_⟩
    exact behavior_of_returns rfl rfl (widthArg_valid idx value)
      (executes_returns (executes_mask idx value))

end Proof
